def sh(sandbox, command: str, timeout: float = 180) -> str:
    """Run a command and return its combined output whatever it exits."""
    try:
        result = sandbox.commands.run(command, timeout=timeout)
        return (result.stdout or "") + (result.stderr or "")
    except Exception as exc:
        return ((getattr(exc, "stdout", "") or "") + (getattr(exc, "stderr", "") or "")
                or f"{type(exc).__name__}: {exc}")

def parse_kv(text: str) -> dict[str, str]:
    out = {}
    for line in text.splitlines():
        if "=" in line:
            k, v = line.split("=", 1)
            out[k.strip()] = v.strip()
    return out

@dataclass(frozen=True)
class Attempt:
    name: str            # "filesystem_escape" | "forbidden_host" | "host_processes"
    command: str         # the exact shell run
    denied: bool         # True == isolation held (the desired outcome)
    signal: str          # the observed proof, e.g. "Permission denied", "exit=1", "Connection refused"
    raw_tail: str        # last ~200 chars of output, redacted

@dataclass(frozen=True)
class IsolationReport:
    region: str
    api_url: str
    domain: str
    template: str
    sdk_version: str
    sandbox_id: str
    started_at: str
    finished_at: str
    attempts: list[Attempt]
    all_denied: bool     # AND over attempts[].denied
    killed: bool
    error: str | None
```

**The three probes (each an `Attempt`):**
1. `filesystem_escape` — try to read a path outside the workspace, e.g. `cat /etc/shadow 2>&1; echo EXIT=$?` and `ls -la / 2>&1 | head; echo EXIT=$?` then attempt a write above the workdir: `touch /flashml_escape_probe 2>&1; echo EXIT=$?`. `denied=True` when the sensitive read is refused / the write fails (non-zero exit or "Permission denied").
2. `forbidden_host` — try to reach the control-plane/host metadata that a user task must never reach, e.g. the cloud metadata endpoint: `curl -sS --max-time 5 http://100.100.100.200/latest/meta-data/ 2>&1; echo EXIT=$?` (Alibaba ECS metadata IP) and a generic egress the workspace should not need. `denied=True` on timeout/refused/non-2xx. **State the signal precisely** — a timeout is different from a refusal; record which.
3. `host_processes` — try to see processes outside the sandbox: `ps aux 2>&1 | wc -l; echo PROCS=$?` and check PID namespace isolation: `ls /proc | grep -c '^[0-9]' ; cat /proc/1/cmdline 2>&1 | tr '\0' ' '`. `denied=True` when only the sandbox's own processes are visible (PID 1 is the sandbox init, not a host process, and the process count is small). This one is a *judgment on the output*, not an exit code — encode the rule explicitly and comment it.

class ModelTransportError(ModelError):         # timeout/network; may_have_applied is meaningless for chat, omit
    ...

        self, messages: list[dict[str, str]], *, model: str, timeout_s: int = DEFAULT_OP_TIMEOUT_S,

    ) -> ModelReply: ...

class QwenModelProvider:                        # the real one, DashScope

```
